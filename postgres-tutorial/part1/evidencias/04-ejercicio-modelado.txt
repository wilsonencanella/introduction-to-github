CREATE TABLE autores (
    id      serial primary key,

CREATE TABLE libros (
    id        serial primary key,
    titulo    varchar(150) NOT NULL,
    anio      int CHECK (anio > 0),
    genero    varchar(50),
    autor_id  int NOT NULL references autores(id)
);

CREATE TABLE socios (
    id      serial primary key,
    nombre  varchar(100) NOT NULL,
    email   varchar(100) UNIQUE
);
CREATE TABLE
CREATE TABLE prestamos (
    id               serial primary key,
    libro_id         int NOT NULL references libros(id),
    socio_id         int NOT NULL references socios(id),
    fecha_prestamo   date NOT NULL,
    fecha_devolucion date   -- si está en null es que todavía no lo devuelven
);

CREATE VIEW prestamos_pendientes AS
    SELECT s.nombre AS socio, l.titulo, p.fecha_prestamo,
           current_date - p.fecha_prestamo AS dias_prestado
        FROM prestamos p
        JOIN socios s ON s.id = p.socio_id
        JOIN libros l ON l.id = p.libro_id
        WHERE p.fecha_devolucion IS NULL;

       rank() OVER (PARTITION BY genero ORDER BY veces DESC) AS puesto
